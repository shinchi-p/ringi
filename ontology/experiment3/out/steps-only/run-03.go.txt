```go
// domain/application/application.go
package application

import (
	"errors"
	"fmt"
	"time"
)

// ApproverID は承認者を識別するID
type ApproverID string

// ApprovalStep は承認ルートの一段を表す
type ApprovalStep struct {
	ApproverID ApproverID
	ApprovedAt *time.Time
}

// IsApproved はこの段が承認済みかどうかを返す
func (s *ApprovalStep) IsApproved() bool {
	return s.ApprovedAt != nil
}

// ApprovalRoute は承認ルート（多段承認の並び）
type ApprovalRoute []*ApprovalStep

// NewApprovalRoute は承認者IDの並びから承認ルートを生成する
func NewApprovalRoute(approverIDs []ApproverID) (ApprovalRoute, error) {
	if len(approverIDs) == 0 {
		return nil, errors.New("承認ルートには少なくとも1名の承認者が必要です")
	}
	route := make(ApprovalRoute, len(approverIDs))
	for i, id := range approverIDs {
		if id == "" {
			return nil, fmt.Errorf("承認者IDが空です (index: %d)", i)
		}
		route[i] = &ApprovalStep{ApproverID: id}
	}
	return route, nil
}

// CurrentStepIndex は現在承認待ちの段のインデックスを返す。
// 全段承認済みの場合は len(route) を返す。
func (r ApprovalRoute) CurrentStepIndex() int {
	for i, step := range r {
		if !step.IsApproved() {
			return i
		}
	}
	return len(r)
}

// IsCompleted は全段が承認済みかどうかを返す
func (r ApprovalRoute) IsCompleted() bool {
	return r.CurrentStepIndex() == len(r)
}

// Reset は承認ルートをリセットする（差戻し時に使用）
func (r ApprovalRoute) Reset() {
	for _, step := range r {
		step.ApprovedAt = nil
	}
}

// ApplicationID は申請を識別するID
type ApplicationID string

// Application は申請を表す集約ルート
type Application struct {
	id            ApplicationID
	status        Status
	approvalRoute ApprovalRoute
	createdAt     time.Time
	updatedAt     time.Time
}

// NewApplication は新しい申請を生成する（初期状態は下書き）
func NewApplication(id ApplicationID, approverIDs []ApproverID) (*Application, error) {
	if id == "" {
		return nil, errors.New("申請IDが空です")
	}
	route, err := NewApprovalRoute(approverIDs)
	if err != nil {
		return nil, fmt.Errorf("承認ルートの生成に失敗しました: %w", err)
	}
	now := time.Now()
	return &Application{
		id:            id,
		status:        StatusDraft,
		approvalRoute: route,
		createdAt:     now,
		updatedAt:     now,
	}, nil
}

// ID は申請IDを返す
func (a *Application) ID() ApplicationID { return a.id }

// Status は現在の状態を返す
func (a *Application) Status() Status { return a.status }

// ApprovalRoute は承認ルートを返す
func (a *Application) ApprovalRoute() ApprovalRoute { return a.approvalRoute }

// CreatedAt は作成日時を返す
func (a *Application) CreatedAt() time.Time { return a.createdAt }

// UpdatedAt は更新日時を返す
func (a *Application) UpdatedAt() time.Time { return a.updatedAt }

// transitionTo は遷移表に従って状態を変更する内部メソッド
func (a *Application) transitionTo(next Status) error {
	if !a.status.CanTransitionTo(next) {
		return fmt.Errorf("状態遷移が許可されていません: %s → %s", a.status, next)
	}
	a.status = next
	a.updatedAt = time.Now()
	return nil
}

// Submit は申請を提出する（下書き → 申請中 / 差戻し → 申請中）
func (a *Application) Submit() error {
	if err := a.transitionTo(StatusSubmitted); err != nil {
		return fmt.Errorf("提出できません: %w", err)
	}
	return nil
}

// Approve は指定した承認者が現在の段を承認する。
// 全段承認が完了した場合、申請は承認済みになる。
func (a *Application) Approve(approverID ApproverID) error {
	if a.status != StatusSubmitted {
		return fmt.Errorf("申請中の申請のみ承認できます (現在の状態: %s)", a.status)
	}

	idx := a.approvalRoute.CurrentStepIndex()
	if idx >= len(a.approvalRoute) {
		// 既に全段承認済みのはずだが念のためガード
		return errors.New("承認可能な段がありません")
	}

	currentStep := a.approvalRoute[idx]
	if currentStep.ApproverID != approverID {
		return fmt.Errorf(
			"承認順序が不正です: 現在の承認者は %s ですが、%s が承認しようとしました",
			currentStep.ApproverID, approverID,
		)
	}

	now := time.Now()
	currentStep.ApprovedAt = &now
	a.updatedAt = now

	// 全段承認済みなら申請を承認済みに遷移
	if a.approvalRoute.IsCompleted() {
		if err := a.transitionTo(StatusApproved); err != nil {
			return fmt.Errorf("承認済みへの遷移に失敗しました: %w", err)
		}
	}

	return nil
}

// Reject は申請を差戻しにする（申請中 → 差戻し）。
// 差戻し時は承認ルートをリセットし、最初の段から承認をやり直す。
func (a *Application) Reject() error {
	if err := a.transitionTo(StatusRejected); err != nil {
		return fmt.Errorf("差戻しできません: %w", err)
	}
	a.approvalRoute.Reset()
	return nil
}
```
