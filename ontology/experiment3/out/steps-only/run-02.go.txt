// domain/application/application.go
package application

import (
	"errors"
	"fmt"
	"time"
)

// ApproverID は承認者を識別するID
type ApproverID string

// ApprovalStep は承認ルートの1段を表す
type ApprovalStep struct {
	ApproverID ApproverID
	ApprovedAt *time.Time
}

// IsApproved はこの段が承認済みかどうかを返す
func (s *ApprovalStep) IsApproved() bool {
	return s.ApprovedAt != nil
}

// ApprovalRoute は承認ルート（承認者の並び）
type ApprovalRoute []*ApprovalStep

// NewApprovalRoute は承認ルートを生成する。承認者は1人以上必要
func NewApprovalRoute(approverIDs []ApproverID) (ApprovalRoute, error) {
	if len(approverIDs) == 0 {
		return nil, errors.New("承認ルートには1人以上の承認者が必要です")
	}
	route := make(ApprovalRoute, len(approverIDs))
	for i, id := range approverIDs {
		route[i] = &ApprovalStep{ApproverID: id}
	}
	return route, nil
}

// CurrentStepIndex は次に承認が必要な段のインデックスを返す。
// すべて承認済みの場合は -1 を返す
func (r ApprovalRoute) CurrentStepIndex() int {
	for i, step := range r {
		if !step.IsApproved() {
			return i
		}
	}
	return -1
}

// AllApproved はすべての段が承認済みかどうかを返す
func (r ApprovalRoute) AllApproved() bool {
	return r.CurrentStepIndex() == -1
}

// ApplicationID は申請を識別するID
type ApplicationID string

// Application は申請を表すエンティティ
type Application struct {
	id     ApplicationID
	status Status
	route  ApprovalRoute
}

// NewApplication は申請を新規作成する。初期状態は下書き
func NewApplication(id ApplicationID, approverIDs []ApproverID) (*Application, error) {
	route, err := NewApprovalRoute(approverIDs)
	if err != nil {
		return nil, err
	}
	return &Application{
		id:     id,
		status: StatusDraft,
		route:  route,
	}, nil
}

// RestoreApplication はDBなどから復元するためのコンストラクタ
func RestoreApplication(id ApplicationID, statusValue string, route ApprovalRoute) (*Application, error) {
	status, err := NewStatus(statusValue)
	if err != nil {
		return nil, err
	}
	return &Application{
		id:     id,
		status: status,
		route:  route,
	}, nil
}

// ID は申請IDを返す
func (a *Application) ID() ApplicationID {
	return a.id
}

// Status は現在の状態を返す
func (a *Application) Status() Status {
	return a.status
}

// Route は承認ルートを返す
func (a *Application) Route() ApprovalRoute {
	return a.route
}

// transitionTo は状態遷移を実行する。遷移表に従わない遷移はエラー
func (a *Application) transitionTo(next Status) error {
	if !a.status.CanTransitionTo(next) {
		return fmt.Errorf("状態 %q から %q への遷移は許可されていません", a.status, next)
	}
	a.status = next
	return nil
}

// Submit は申請を提出する（下書き→申請中、差戻し→申請中）
func (a *Application) Submit() error {
	if err := a.transitionTo(StatusSubmitted); err != nil {
		return err
	}
	// 再提出時は承認済みステップをリセットする
	for _, step := range a.route {
		step.ApprovedAt = nil
	}
	return nil
}

// Approve は指定した承認者が現在の段を承認する。
// すべての段が承認されると申請は承認済みになる
func (a *Application) Approve(approverID ApproverID) error {
	if a.status != StatusSubmitted {
		return fmt.Errorf("申請中の状態でなければ承認できません（現在: %q）", a.status)
	}

	idx := a.route.CurrentStepIndex()
	if idx == -1 {
		return errors.New("すべての承認段がすでに承認済みです")
	}

	step := a.route[idx]
	if step.ApproverID != approverID {
		return fmt.Errorf("現在の承認段の承認者は %q ですが、%q が承認しようとしています", step.ApproverID, approverID)
	}

	now := time.Now()
	step.ApprovedAt = &now

	// すべての段が承認されたら承認済みに遷移
	if a.route.AllApproved() {
		if err := a.transitionTo(StatusApproved); err != nil {
			return err
		}
	}

	return nil
}

// Reject は申請を差戻しにする（申請中→差戻し）
func (a *Application) Reject() error {
	return a.transitionTo(StatusRejected)
}
